-- =============================================
-- PROCEDIMIENTO 5: RESUMEN COMPLETO DEL DASHBOARD (CORREGIDO)
-- =============================================
CREATE OR REPLACE PROCEDURE sp_resumen_dashboard(
    IN p_start_date TIMESTAMPTZ,
    IN p_end_date TIMESTAMPTZ,
    IN p_search_query VARCHAR DEFAULT '',
    IN p_ids_pedidos INTEGER[] DEFAULT NULL
)
LANGUAGE plpgsql
AS $$
DECLARE
    v_ventas_totales DECIMAL;
    v_total_pedidos INTEGER;
    v_promedio_venta DECIMAL;
    v_total_mesas INTEGER;
    v_platillos_disponibles INTEGER;
BEGIN
    -- 1. Calcular métricas según el caso
    IF p_ids_pedidos IS NOT NULL AND array_length(p_ids_pedidos, 1) > 0 THEN
        SELECT 
            COALESCE(SUM("montoTotal") * 1.15, 0),
            COUNT(*),
            CASE WHEN COUNT(*) > 0 THEN ROUND(AVG("montoTotal") * 1.15, 2) ELSE 0 END
        INTO v_ventas_totales, v_total_pedidos, v_promedio_venta
        FROM "Pedido"
        WHERE "idPedido" = ANY(p_ids_pedidos)
            AND "estado_factura" = 'VIGENTE'; -- Se excluyen anuladas por IDs específicos
    ELSE
        SELECT 
            COALESCE(SUM("montoTotal") * 1.15, 0),
            COUNT(*),
            CASE WHEN COUNT(*) > 0 THEN ROUND(AVG("montoTotal") * 1.15, 2) ELSE 0 END
        INTO v_ventas_totales, v_total_pedidos, v_promedio_venta
        FROM "Pedido"
        WHERE fecha >= p_start_date
            AND fecha <= p_end_date
            AND "estadoDePago" = TRUE
            AND "estado_factura" = 'VIGENTE'; -- Se excluyen anuladas del cálculo general
    END IF;
    
    -- 2. Conteos generales
    SELECT COUNT(*) INTO v_total_mesas FROM "Mesa";
    SELECT COUNT(*) INTO v_platillos_disponibles FROM "ProductoMenu" WHERE disponible = TRUE;
    
    -- 3. Guardar métricas en tabla temporal
    CREATE TEMP TABLE IF NOT EXISTS temp_metricas_dashboard (
        ventas_totales DECIMAL,
        total_pedidos INTEGER,
        promedio_venta DECIMAL,
        total_mesas INTEGER,
        platillos_disponibles INTEGER
    ) ON COMMIT DROP;
    
    DELETE FROM temp_metricas_dashboard;
    INSERT INTO temp_metricas_dashboard VALUES (
        v_ventas_totales, v_total_pedidos, v_promedio_venta, 
        v_total_mesas, v_platillos_disponibles
    );
    
    -- 4. Guardar top productos en tabla temporal
    CREATE TEMP TABLE IF NOT EXISTS temp_top_productos (
        nombre_producto VARCHAR,
        cantidad_total BIGINT,
        ingresos_totales DECIMAL,
        porcentaje_ventas DECIMAL
    ) ON COMMIT DROP;
    
    DELETE FROM temp_top_productos;
    
    INSERT INTO temp_top_productos
    WITH ventas_filtradas AS (
        SELECT 
            pr.nombre,
            ppm.cantidad,
            pr.precio
        FROM "Pedido_ProductoMenu" ppm
        INNER JOIN "ProductoMenu" pr ON ppm."idProductoMenu_id" = pr."idProductoMenu"
        INNER JOIN "Pedido" pm ON ppm."idPedido_id" = pm."idPedido"
        WHERE pm."estadoDePago" = TRUE
            AND pm."estado_factura" = 'VIGENTE' -- Se excluyen anuladas del top de productos
            AND (
                (p_ids_pedidos IS NULL AND pm.fecha >= p_start_date AND pm.fecha <= p_end_date)
                OR (p_ids_pedidos IS NOT NULL AND pm."idPedido" = ANY(p_ids_pedidos))
            )
    ),
    productos_agrupados AS (
        SELECT 
            nombre,
            SUM(cantidad) AS cantidad_total,
            SUM(cantidad * precio) AS ingresos_totales,
            SUM(SUM(cantidad)) OVER() AS total_general
        FROM ventas_filtradas
        GROUP BY nombre
    )
    SELECT 
        nombre,
        cantidad_total,
        ingresos_totales,
        ROUND((cantidad_total::DECIMAL / NULLIF(total_general, 0) * 100), 2)
    FROM productos_agrupados
    ORDER BY cantidad_total DESC
    LIMIT 5;
    
    -- 5. Guardar top mesas en tabla temporal
    CREATE TEMP TABLE IF NOT EXISTS temp_top_mesas (
        mesa_id INTEGER,
        numero_mesa INTEGER,
        total_pedidos BIGINT,
        total_ventas DECIMAL,
        promedio_por_pedido DECIMAL,
        clientes_unicos BIGINT
    ) ON COMMIT DROP;
    
    DELETE FROM temp_top_mesas;
    
    INSERT INTO temp_top_mesas
    SELECT 
        m."idMesa",
        m."idMesa", 
        COUNT(p."idPedido") AS total_pedidos,
        COALESCE(SUM(p."montoTotal") * 1.15, 0),
        ROUND(AVG(p."montoTotal") * 1.15, 2),
        COUNT(DISTINCT p."idCliente_id")
    FROM "Pedido" p
    INNER JOIN "Mesa" m ON p."idMesa_id" = m."idMesa"
    WHERE p."estadoDePago" = TRUE
        AND p."estado_factura" = 'VIGENTE'
        AND p."idMesa_id" IS NOT NULL
        AND (
            (p_ids_pedidos IS NULL AND p.fecha >= p_start_date AND p.fecha <= p_end_date)
            OR (p_ids_pedidos IS NOT NULL AND p."idPedido" = ANY(p_ids_pedidos))
        )
    GROUP BY m."idMesa"
    ORDER BY total_pedidos DESC
    LIMIT 10;
    
    -- 6. Si hay búsqueda, guardar resultados en tabla temporal
    IF p_search_query != '' THEN
        CREATE TEMP TABLE IF NOT EXISTS temp_busqueda_clientes (
            id_pedido INTEGER,
            nombre_cliente VARCHAR,
            mesa_numero INTEGER,
            fecha_pedido TIMESTAMPTZ,
            monto_total DECIMAL,
            estado_pago BOOLEAN,
            metodo_pago VARCHAR,
            total_encontrados BIGINT
        ) ON COMMIT DROP;
        
        DELETE FROM temp_busqueda_clientes;
        
        INSERT INTO temp_busqueda_clientes
        SELECT 
            p."idPedido",
            c.nombre,
            m."idMesa",
            p.fecha,
            (p."montoTotal" * 1.15)::DECIMAL,
            p."estadoDePago",
            p."metodoPago",
            COUNT(*) OVER()
        FROM "Pedido" p
        INNER JOIN "Cliente" c ON p."idCliente_id" = c."idCliente"
        LEFT JOIN "Mesa" m ON p."idMesa_id" = m."idMesa"
        WHERE p."estadoDePago" = TRUE
            AND p."estado_factura" = 'VIGENTE'
            AND p.fecha >= p_start_date
            AND p.fecha <= p_end_date
            AND c.nombre ILIKE '%' || p_search_query || '%'
        ORDER BY p.fecha DESC
        LIMIT 50;
    END IF;
    
END;
$$;
